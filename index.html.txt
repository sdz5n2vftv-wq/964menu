    <script>
        const cart = {};
        function changeQty(name, price, delta) {
            if (!cart[name]) cart[name] = { price: price, qty: 0 };
            cart[name].qty += delta;
            if (cart[name].qty < 0) cart[name].qty = 0;
            document.getElementById('qty-' + name).innerText = cart[name].qty;
        }

        function sendOrder() {
            let message = "مرحباً، أود طلب الأتي من 964 Street Bites:%0A";
            let total = 0;
            let hasItems = false;

            for (let name in cart) {
                if (cart[name].qty > 0) {
                    hasItems = true;
                    let itemTotal = cart[name].price * cart[name].qty;
                    total += itemTotal;
                    message += `- ${name} (عدد: ${cart[name].qty}) - ${itemTotal} د.ع%0A`;
                }
            }

            if (!hasItems) {
                alert("الرجاء اختيار وجبة واحدة على الأقل قبل إرسال الطلب!");
                return;
            }

            message += `%0Aالمجموع الكلي: ${total} د.ع`;
            let phone = "9647722202416";
            window.open(`https://wa.me/${phone}?text=${message}`, '_blank');
        }
    </script>
